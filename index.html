const SCRIPT_URL = 'URL_WEB_APP_ANDA_DISINI'; window.onload = function() { fetch(SCRIPT_URL) .then(res => res.json()) .then(data => { // Data asli mulai dari baris ke-5 (indeks 4) const isiData = data.slice(4); console.log("Data berhasil ditarik:", isiData); // Hitung total dan isi dashboard let totalAnggaran = 0; isiData.forEach(row => { // row[5] adalah kolom Anggaran, row[6] adalah kolom Realisasi totalAnggaran += parseFloat(row[5].replace(/[^0-9]/g, '')) || 0; }); // Update tampilan (Pastikan ID di HTML sama persis) document.getElementById('cardTotalAnggaran').innerText = "Rp " + totalAnggaran.toLocaleString('id-ID'); }) .catch(err => alert("Gagal mengambil data: " + err)); };
